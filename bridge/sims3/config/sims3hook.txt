# The Sims 3 camera hook: options read once at start (key = value, integers).
# terrain       1 = terrain through the runtime's terrain baker (0 = the pre-M17 capture)
# ringTrace     1 = keep a rolling trace of every draw and event of the last ~300 frames; the mark key writes it to the log (diagnostic; 0 = off)
# markKey       virtual-key code of the mark key (F9 and backtick always work); 45 = Insert, 220 = backslash, 36 = Home, 35 = End, 19 = Pause, 145 = Scroll Lock
# exposureFromLight  1 = the runtime's sky brightness stays 1 (the sky the game draws lights the scene as it is) and the ceiling of
#               its auto-exposure (rtx.autoExposure.evMaxValue) follows the luminance of the sun and the moon as sent, relative to the
#               game's full daylight; needs exposeRemixApi = True
# dayEvMax / nightEvMax  the auto-exposure ceiling in hundredths of an EV at full day and at no light (runtime default 500; 0 at no light since 2026-09-30: no brightening of the night)
terrain = 1
ringTrace = 0
markKey = 220
# The lights go to the runtime through the Remix API (exposeRemixApi = True in .trex\bridge.conf). The lamps are the
# game's own: their lights and shapes from sims3lights.txt next to the DLL (sims3/tools/lite_table.py writes it).
# The sun and the moon are the game's one directional light, as its terrain shaders are handed it, told apart by the
# game's clock; the moon by its share, the dawn eased, the sun with an afterglow at dusk.
# sunAngle      the sun's and the moon's angular diameter in thousandths of a degree
# sunRadiance   their radiance per unit of the game's colour, thousandths
# moonLight     the moon's share of the game's moonlight in percent (100 = the game's own; 0 = none, the night stays dark)
# Dusk and dawn are cross-fades of the two lights in equal parts, the hook's own (the game's own changes are over in twelve minutes):
# dawnMinutes   from sunrise, over this many game minutes, the moon fades from its floor to nothing while the sun rises to the game's own light; 0 = as the game has it
# duskMinutes   from the moment the game's light becomes the moon's (an hour after sunset), over this many game minutes, the sun's afterglow fades
#               to nothing while the moon rises to its floor; 0 = as the game has it
# duskLevel     the level of full sunlight, in thousandths, below which the sun is held as the afterglow until the moon comes
# lampRadius    the lamps' sphere radius in thousandths of a unit
# lampRadiance  the lamps' radiance per unit of colour, thousandths
exposureFromLight = 1
dayEvMax = 500
nightEvMax = 0
sunAngle = 2000
sunRadiance = 1000
moonLight = 2
dawnMinutes = 60
duskMinutes = 60
duskLevel = 100
lampRadius = 150
lampRadiance = 40000
# The lamps' shapes come from the game's light definitions (sims3lights.txt, format 2).
# lampShapes        1 = spots as cones, lamp shades as two cones plus the light through the shade, tubes as cylinders; 0 = plain spheres
# lampConeScale     every cone angle times this, in thousandths (1000 = the definitions' angles taken as the angle from the axis; 500 = half as wide)
# lampConeSoftness  the softness of a cone's edge, in thousandths (0 = hard, 1000 = soft across the whole cone)
# lampShadeGlow     the light through a lamp shade times this, in thousandths (0 = none)
# Which objects are lamps, where they stand, whether they are on, in which colour and at which level comes from the
# lamp reporter, a script mod (Sims3RtxLamps.package in Mods\Packages; sims3/scriptmod). Without it no lamp gives light.
# lampMax           the most lamps lit at once, the nearest to the camera's target first (1..96)
# lampWorldLights   1 = the world lights too (a street lamp's), faded in and out by the game's own night switch; 0 = they stay dark
lampMax = 96
lampWorldLights = 1
lampShapes = 1
lampConeScale = 1000
lampConeSoftness = 300
lampShadeGlow = 1000
