# The Sims 3 camera hook: options read once at start (key = value, integers).
# terrain       1 = terrain through the runtime's terrain baker (0 = the pre-M17 capture)
# ringTrace     1 = keep a rolling trace of every draw and event of the last ~300 frames; the mark key writes it to the log (diagnostic; 0 = off)
# markKey       virtual-key code of the mark key (F9 and backtick always work); 45 = Insert, 220 = backslash, 36 = Home, 35 = End, 19 = Pause, 145 = Scroll Lock
# skyFromSun    1 = night from the sun: the runtime's sky brightness (rtx.skyBrightness) and the ceiling of its auto-exposure
#               (rtx.autoExposure.evMaxValue) follow the luminance of the sun the hook holds; needs exposeRemixApi = True
# skyDayLevel   the sun luminance of full day, in thousandths (the reference; a higher one seen in the run raises it)
# skyMinBrightness  the darkest sky brightness, in thousandths
# dayEvMax / nightEvMax  the auto-exposure ceiling in hundredths of an EV at full day and at the floor (runtime default 500)
terrain = 1
ringTrace = 0
markKey = 220
# The lights go to the runtime through the Remix API (exposeRemixApi = True in .trex\bridge.conf); the lamps are the
# game's own, from sims3lights.txt next to the DLL (sims3/tools/lite_table.py --hook writes it).
# sunAngle      the sun's angular diameter in thousandths of a degree
# sunRadiance   the sun's radiance per unit of rig colour, thousandths
# lampRadius    the lamps' sphere radius in thousandths of a unit
# lampRadiance  the lamps' radiance per unit of colour, thousandths
# lampOffTexture the content hash (hex) of the lot's shared light map, bound at a lamp's shader stage 2 while the lamp is OFF
#               (f3e6c2b1f83e7fd8 on the current game version). A lamp drawn with it, or with a light rig that is not all
#               zeros, is off; drawn with its own glow map and a rig of zeros, it is on. The "lamp object at the mark" log
#               lines (backslash) show both for any lamp in view.
skyFromSun = 1
skyDayLevel = 900
skyMinBrightness = 30
dayEvMax = 500
nightEvMax = 100
sunAngle = 2000
sunRadiance = 1000
lampRadius = 150
lampRadiance = 40000
lampOffTexture = f3e6c2b1f83e7fd8
